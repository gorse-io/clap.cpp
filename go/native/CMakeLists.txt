cmake_minimum_required(VERSION 3.20)
project(clap_cpp_go VERSION 0.1.0 LANGUAGES C CXX)
if(NOT UNIX)
    message(FATAL_ERROR "Go bindings currently support Linux and macOS")
endif()
get_filename_component(CLAP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(source ggml/CMakeLists.txt thirdparty/json/CMakeLists.txt thirdparty/icu/icu4c/source/common/unicode/utypes.h thirdparty/miniaudio/miniaudio.h)
    if(NOT EXISTS "${CLAP_ROOT}/${source}")
        message(FATAL_ERROR "Native sources missing: run git submodule update --init --recursive")
    endif()
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_STATIC OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
foreach(backend METAL CUDA HIP VULKAN SYCL OPENCL OPENVINO RPC MUSA WEBGPU ZDNN CANN
                VIRTGPU VIRTGPU_BACKEND ET HEXAGON ZENDNN BLAS ACCELERATE)
    set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(feature BACKEND_DL OPENMP BUILD_TESTS BUILD_EXAMPLES)
    set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(isa SSE42 AVX AVX2 AVX_VNNI BMI2 FMA F16C AVX512 AVX512_VBMI
            AVX512_VNNI AVX512_BF16 AMX_TILE AMX_INT8 AMX_BF16)
    set(GGML_${isa} OFF CACHE BOOL "" FORCE)
endforeach()
set(CLAP_USE_SYSTEM_ICU OFF CACHE BOOL "" FORCE)
add_subdirectory("${CLAP_ROOT}/ggml" ggml EXCLUDE_FROM_ALL)
add_subdirectory("${CLAP_ROOT}/thirdparty/json" json EXCLUDE_FROM_ALL)
include("${CLAP_ROOT}/cmake/icu.cmake")
add_subdirectory("${CLAP_ROOT}/src" core EXCLUDE_FROM_ALL)
add_library(clap-go SHARED bridge.cpp)
target_link_libraries(clap-go PRIVATE clap)
target_compile_features(clap-go PRIVATE cxx_std_17)
# The installed bridge contains the static core, ggml and bundled ICU; only
# ordinary platform runtime libraries remain dynamic dependencies.
set_target_properties(clap-go PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
install(TARGETS clap-go LIBRARY DESTINATION lib COMPONENT go)
install(FILES bridge.h DESTINATION include/clap-go COMPONENT go)
configure_file(clap-go.pc.in clap-go.pc @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/clap-go.pc" DESTINATION lib/pkgconfig COMPONENT go)
