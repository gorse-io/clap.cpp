cmake_minimum_required(VERSION 3.20)
# Script mode runs after generation: the File API contains CMake's fully
# resolved, ordered transitive link closure (including SDK imported targets).
# Do not duplicate that graph in build.rs or guess SDK library directories.
if(DEFINED CLAP_RUST_LINK_MANIFEST)
    include("${CLAP_RUST_BUILD}/cargo-link-config.cmake")
    file(GLOB indexes "${CLAP_RUST_BUILD}/.cmake/api/v1/reply/index-*.json")
    list(SORT indexes ORDER DESCENDING)
    list(GET indexes 0 index)
    file(READ "${index}" index_json)
    string(JSON model_file GET "${index_json}" reply codemodel-v2 jsonFile)
    get_filename_component(reply_dir "${index}" DIRECTORY)
    file(READ "${reply_dir}/${model_file}" model)
    string(JSON count LENGTH "${model}" configurations)
    math(EXPR last "${count} - 1")
    foreach(config RANGE ${last})
        string(JSON name GET "${model}" configurations ${config} name)
        if(name STREQUAL "Release" OR count EQUAL 1)
            set(selected ${config})
        endif()
    endforeach()
    string(JSON count LENGTH "${model}" configurations ${selected} targets)
    math(EXPR last "${count} - 1")
    foreach(i RANGE ${last})
        string(JSON name GET "${model}" configurations ${selected} targets ${i} name)
        if(name STREQUAL "clap-rust-link-probe")
            string(JSON target_file GET "${model}" configurations ${selected} targets ${i} jsonFile)
        endif()
    endforeach()
    file(READ "${reply_dir}/${target_file}" target_json)
    string(JSON count LENGTH "${target_json}" link commandFragments)
    math(EXPR last "${count} - 1")
    set(directives "")
    foreach(i RANGE ${last})
        string(JSON role GET "${target_json}" link commandFragments ${i} role)
        string(JSON fragment GET "${target_json}" link commandFragments ${i} fragment)
        # Compiler driver defaults and build-tree rpaths are not portable Cargo
        # metadata. Shared SDKs must be available to the system runtime loader.
        if(NOT role MATCHES "^(libraries|libraryPath|frameworkPath)$")
            continue()
        endif()
        if(CLAP_RUST_WINDOWS)
            separate_arguments(items WINDOWS_COMMAND "${fragment}")
        else()
            separate_arguments(items UNIX_COMMAND "${fragment}")
        endif()
        set(framework_next OFF)
        foreach(item IN LISTS items)
            if(framework_next)
                string(APPEND directives "cargo:rustc-link-lib=framework=${item}\n")
                set(framework_next OFF)
            elseif(item STREQUAL "-framework" OR item STREQUAL "-weak_framework")
                set(framework_next ON)
            elseif(item MATCHES "^-L(.+)" OR item MATCHES "^/LIBPATH:(.+)")
                string(APPEND directives "cargo:rustc-link-search=native=${CMAKE_MATCH_1}\n")
            elseif(item MATCHES "^-F(.+)")
                string(APPEND directives "cargo:rustc-link-search=framework=${CMAKE_MATCH_1}\n")
            elseif(item MATCHES "^-Wl,-rpath(-link)?,")
                # CMake places generated rpaths in the libraries role too.
                # Do not export ephemeral build-tree or SDK loader policy.
                continue()
            elseif(item MATCHES "^-l(.+)")
                string(APPEND directives "cargo:rustc-link-lib=${CMAKE_MATCH_1}\n")
            elseif(item STREQUAL "-pthread")
                string(APPEND directives "cargo:rustc-link-lib=pthread\n")
            elseif(item MATCHES "(^|/)([^/]+)\\.framework(/.*)?$")
                string(APPEND directives "cargo:rustc-link-lib=framework=${CMAKE_MATCH_2}\n")
            elseif(item MATCHES "\\.(a|lib|dylib|tbd)$" OR item MATCHES "\\.so(\\.[0-9.]+)?$")
                get_filename_component(path "${item}" ABSOLUTE BASE_DIR "${CLAP_RUST_BUILD}")
                get_filename_component(dir "${path}" DIRECTORY)
                get_filename_component(filename "${path}" NAME)
                string(REGEX REPLACE "\\.(a|lib|dylib|tbd)$" "" stem "${filename}")
                if(NOT item MATCHES "\\.lib$")
                    string(REGEX REPLACE "^lib" "" stem "${stem}")
                endif()
                if(stem IN_LIST CLAP_RUST_ARCHIVES)
                    string(APPEND directives "cargo:rustc-link-lib=static=${stem}\n")
                else()
                    string(APPEND directives "cargo:rustc-link-search=native=${dir}\n")
                    if(item MATCHES "\\.a$")
                        string(APPEND directives "cargo:rustc-link-lib=static:+verbatim=${filename}\n")
                    elseif(CLAP_RUST_APPLE OR CLAP_RUST_WINDOWS)
                        string(APPEND directives "cargo:rustc-link-lib=dylib=${stem}\n")
                    else()
                        string(APPEND directives "cargo:rustc-link-lib=dylib:+verbatim=${filename}\n")
                    endif()
                endif()
            elseif(item MATCHES "^[A-Za-z0-9_+.-]+$")
                string(REGEX REPLACE "\\.lib$" "" item "${item}")
                string(APPEND directives "cargo:rustc-link-lib=${item}\n")
            else()
                message(FATAL_ERROR "Unsupported native link item for Cargo: ${item}")
            endif()
        endforeach()
    endforeach()
    file(WRITE "${CLAP_RUST_LINK_MANIFEST}" "${directives}")
    return()
endif()

# Keep native archives static and the no-feature build portable. GPU SDK
# runtimes are discovered by CMake and can be shared (GGML_STATIC is different
# from BUILD_SHARED_LIBS and is explicitly unsupported by ggml's HIP backend).
set(GGML_STATIC OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
foreach(backend METAL CUDA HIP VULKAN SYCL OPENCL OPENVINO RPC MUSA WEBGPU ZDNN CANN
                VIRTGPU VIRTGPU_BACKEND ET HEXAGON ZENDNN BLAS ACCELERATE)
    set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(backend CUDA VULKAN HIP METAL)
    option(CLAP_RUST_${backend} "Enable the ${backend} backend for Rust" OFF)
    set(GGML_${backend} ${CLAP_RUST_${backend}} CACHE BOOL "" FORCE)
endforeach()
# Embed Metal shader sources in the archive; never depend on build-tree assets.
set(GGML_METAL_EMBED_LIBRARY ${CLAP_RUST_METAL} CACHE BOOL "" FORCE)
if(CLAP_RUST_CUDA AND CLAP_RUST_HIP)
    message(FATAL_ERROR "cuda and rocm cannot be enabled together (shared ggml CUDA symbols)")
endif()
if(CLAP_RUST_METAL AND NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    message(FATAL_ERROR "The metal Cargo feature requires macOS")
endif()
if(CLAP_RUST_HIP AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The rocm Cargo feature currently requires Linux")
endif()
foreach(feature BACKEND_DL OPENMP BUILD_TESTS BUILD_EXAMPLES)
    set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
# Avoid host-specific x86 ISA requirements in downstream executables.
foreach(isa SSE42 AVX AVX2 AVX_VNNI BMI2 FMA F16C AVX512 AVX512_VBMI
            AVX512_VNNI AVX512_BF16 AMX_TILE AMX_INT8 AMX_BF16)
    set(GGML_${isa} OFF CACHE BOOL "" FORCE)
endforeach()
set(CLAP_USE_SYSTEM_ICU OFF CACHE BOOL "" FORCE)
add_subdirectory("${CLAP_ROOT}/ggml" ggml EXCLUDE_FROM_ALL)
add_subdirectory("${CLAP_ROOT}/thirdparty/json" json EXCLUDE_FROM_ALL)
include("${CLAP_ROOT}/cmake/icu.cmake")
add_subdirectory("${CLAP_ROOT}/src" core EXCLUDE_FROM_ALL)

add_library(clap-rust-native STATIC bridge.cpp)
target_link_libraries(clap-rust-native PRIVATE clap)
target_compile_features(clap-rust-native PRIVATE cxx_std_17)
set(rust_archives clap-rust-native clap ggml ggml-cpu)
foreach(backend cuda vulkan hip metal)
    if(TARGET ggml-${backend})
        list(APPEND rust_archives ggml-${backend})
    endif()
endforeach()
list(APPEND rust_archives ggml-base clap-icu-i18n clap-icu-uc clap-icu-data)
install(TARGETS ${rust_archives} ARCHIVE DESTINATION lib COMPONENT rust)

# transitive library closure, including dependencies of imported SDK targets.
