cmake_minimum_required(VERSION 3.20)
project(clap_cpp_python VERSION 0.1.0 LANGUAGES C CXX)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/native/include/clap.h")
    set(CLAP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/native")
else()
    get_filename_component(CLAP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
endif()
foreach(source ggml/CMakeLists.txt thirdparty/json/CMakeLists.txt thirdparty/miniaudio/miniaudio.h)
    if(NOT EXISTS "${CLAP_ROOT}/${source}")
        message(FATAL_ERROR "Native sources missing: run git submodule update --init --recursive")
    endif()
endforeach()

# Each wheel contains one statically linked runtime. GPU wheels install into
# separate packages so extras never overwrite the base package's CPU extension.
set(CLAP_PYTHON_BACKEND "CPU" CACHE STRING "Python wheel backend: CPU, CUDA, HIP or VULKAN")
set_property(CACHE CLAP_PYTHON_BACKEND PROPERTY STRINGS CPU CUDA HIP VULKAN)
if(NOT CLAP_PYTHON_BACKEND MATCHES "^(CPU|CUDA|HIP|VULKAN)$")
    message(FATAL_ERROR "Unknown Python backend: ${CLAP_PYTHON_BACKEND}")
endif()
set(CLAP_PYTHON_PACKAGE "clap_cpp")
if(NOT CLAP_PYTHON_BACKEND STREQUAL "CPU")
    string(TOLOWER "${CLAP_PYTHON_BACKEND}" backend_name)
    if(backend_name STREQUAL "hip")
        set(backend_name "rocm")
    endif()
    set(CLAP_PYTHON_PACKAGE "clap_cpp_${backend_name}")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
foreach(backend METAL CUDA HIP VULKAN SYCL OPENCL OPENVINO RPC MUSA WEBGPU ZDNN
                VIRTGPU VIRTGPU_BACKEND ET HEXAGON ZENDNN BLAS ACCELERATE)
    set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
if(NOT CLAP_PYTHON_BACKEND STREQUAL "CPU")
    set(GGML_${CLAP_PYTHON_BACKEND} ON CACHE BOOL "" FORCE)
endif()
foreach(feature BACKEND_DL OPENMP BUILD_TESTS BUILD_EXAMPLES)
    set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
# Avoid host-specific x86 ISA requirements in redistributed wheels.
foreach(isa SSE42 AVX AVX2 AVX_VNNI BMI2 FMA F16C AVX512 AVX512_VBMI
            AVX512_VNNI AVX512_BF16 AMX_TILE AMX_INT8 AMX_BF16)
    set(GGML_${isa} OFF CACHE BOOL "" FORCE)
endforeach()
set(CLAP_USE_SYSTEM_ICU OFF CACHE BOOL "" FORCE)
add_subdirectory("${CLAP_ROOT}/ggml" ggml EXCLUDE_FROM_ALL)
add_subdirectory("${CLAP_ROOT}/thirdparty/json" json EXCLUDE_FROM_ALL)
include("${CLAP_ROOT}/cmake/icu.cmake")
add_subdirectory("${CLAP_ROOT}/src" core EXCLUDE_FROM_ALL)

find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_clap_cpp MODULE bindings.cpp)
target_link_libraries(_clap_cpp PRIVATE clap)
target_compile_features(_clap_cpp PRIVATE cxx_std_17)
set_target_properties(_clap_cpp PROPERTIES CXX_EXTENSIONS OFF)
install(TARGETS _clap_cpp LIBRARY DESTINATION ${CLAP_PYTHON_PACKAGE} COMPONENT python
                          RUNTIME DESTINATION ${CLAP_PYTHON_PACKAGE} COMPONENT python)
install(FILES "${CLAP_ROOT}/ggml/LICENSE" DESTINATION ${CLAP_PYTHON_PACKAGE}/licenses RENAME ggml.txt COMPONENT python)
install(FILES "${CLAP_ROOT}/thirdparty/miniaudio/LICENSE" DESTINATION ${CLAP_PYTHON_PACKAGE}/licenses RENAME miniaudio.txt COMPONENT python)
install(FILES "${CLAP_ROOT}/thirdparty/icu/LICENSE" DESTINATION ${CLAP_PYTHON_PACKAGE}/licenses RENAME icu.txt COMPONENT python)
install(FILES "${CLAP_ROOT}/thirdparty/json/LICENSE.MIT" DESTINATION ${CLAP_PYTHON_PACKAGE}/licenses RENAME json.txt COMPONENT python)
